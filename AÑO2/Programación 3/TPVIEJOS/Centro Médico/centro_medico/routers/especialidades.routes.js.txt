import express from "express";
const router = express.Router();

import { pool } from "../db/conexion.js";
import { check, param } from "express-validator";
import { validarCampos } from "../middlewares/validarCampos.js";


// 🟢 ADD
router.post('/',
[
    check('nombre').notEmpty(),
    validarCampos
],
async (req, res) => {

    const { nombre } = req.body;

    const [result] = await pool.execute(
        'INSERT INTO especialidades (nombre) VALUES (?)',
        [nombre]
    );

    res.send({ id: result.insertId });
});


// 🔵 BROWSE
router.get('/', async (req, res) => {

    const [data] = await pool.query(
        'SELECT * FROM especialidades WHERE activo = 1'
    );

    res.send(data);
});


// 🟣 READ
router.get('/:id',
[
    param('id').isInt(),
    validarCampos
],
async (req, res) => {

    const [data] = await pool.execute(
        'SELECT * FROM especialidades WHERE id_especialidad = ?',
        [req.params.id]
    );

    res.send(data);
});


// 🟡 UPDATE
router.put('/:id',
[
    param('id').isInt(),
    validarCampos
],
async (req, res) => {

    await pool.execute(
        'UPDATE especialidades SET nombre = ? WHERE id_especialidad = ?',
        [req.body.nombre, req.params.id]
    );

    res.send("Actualizado");
});


// 🔴 DELETE (lógico)
router.delete('/:id',
[
    param('id').isInt(),
    validarCampos
],
async (req, res) => {

    await pool.execute(
        'UPDATE especialidades SET activo = 0 WHERE id_especialidad = ?',
        [req.params.id]
    );

    res.send("Eliminado");
});

export default router;
